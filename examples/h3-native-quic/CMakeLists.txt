cmake_minimum_required(VERSION 3.16)
project(h3_native_quic LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "h3-native-quic requires Linux")
endif()
if(NOT EXISTS "${PICOQUIC_SOURCE_DIR}/picoquic/picoquic.h")
    message(FATAL_ERROR "Set PICOQUIC_SOURCE_DIR to the pinned offline source")
endif()
if(NOT EXISTS "${FETCHCONTENT_SOURCE_DIR_PICOTLS}/include/picotls.h")
    message(FATAL_ERROR "Set FETCHCONTENT_SOURCE_DIR_PICOTLS to the pinned offline source")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_SHARED_LIBS OFF)
set(PICOQUIC_FETCH_PTLS ON CACHE BOOL "" FORCE)
set(PTLS_WITH_FUSION OFF CACHE BOOL "" FORCE)
set(BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(BUILD_LOGLIB ON CACHE BOOL "" FORCE)
add_subdirectory("${PICOQUIC_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/picoquic" EXCLUDE_FROM_ALL)
set(RPC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../ggml/src/ggml-rpc")
add_library(h3-native-quic SHARED h3-quic.cpp "${RPC_DIR}/transport-quic.cpp")
target_include_directories(h3-native-quic PRIVATE "${RPC_DIR}" "${RPC_DIR}/.." "${RPC_DIR}/../../include")
target_compile_definitions(h3-native-quic PRIVATE GGML_RPC_QUIC)
target_link_libraries(h3-native-quic PRIVATE picoquic-core picoquic-log)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/h3-native-quic.exports" "{ global: h3q_*; local: *; };\n")
target_link_options(h3-native-quic PRIVATE "LINKER:--exclude-libs,ALL" "LINKER:--version-script=${CMAKE_CURRENT_BINARY_DIR}/h3-native-quic.exports")
set_target_properties(h3-native-quic PROPERTIES BUILD_RPATH_USE_ORIGIN ON)
